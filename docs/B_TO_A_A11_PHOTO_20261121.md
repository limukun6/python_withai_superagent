```sh
git bundle verify BUNDLE
git fetch BUNDLE refs/heads/codex/b-a11-integration:refs/heads/codex/b-a11-integration
git switch codex/b-a11-integration
```

python -m scripts.run_b_photo_pilot --source FLOWERS_TGZ --weights MOBILENET_PTH --output PHOTO_PRE --prepare-only
